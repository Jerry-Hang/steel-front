
        // ---- 字体图集（中文/西文动态文字）：128x128 R8G8B8A8，每格 8x8 -> 16x16 字符 ----
        const FONT_CELL: u32 = 8;
        const FONT_ATLAS_SIZE: u32 = 128;
        let atlas_pixels = (FONT_ATLAS_SIZE * FONT_ATLAS_SIZE) as usize;
        self.hud_font_bitmap = vec![0u8; atlas_pixels * 4];
        // ASCII 0x20..0x7E 预注册（格子 0..95），CJK 从 96 起动态分配
        for (i, ch) in (0x20u32..=0x7Eu32).enumerate() {
            self.hud_font_glyphs.insert(char::from_u32(ch).unwrap(), i as u8);
        }
        // 创建图集图像（一次分配，字形逐格绘制后上传）
        let (img, mem) = self.create_image(
            FONT_ATLAS_SIZE,
            FONT_ATLAS_SIZE,
            vk::Format::R8G8B8A8_SRGB,
            vk::ImageUsageFlags::SAMPLED | vk::ImageUsageFlags::TRANSFER_DST,
        )?;
        self.hud_font_image = img;
        self.hud_font_memory = mem;
        self.hud_font_image_view = self.create_image_view(
            img,
            vk::Format::R8G8B8A8_SRGB,
            vk::ImageAspectFlags::COLOR,
        )?;
        let sampler_info = vk::SamplerCreateInfo::default()
            .mag_filter(vk::Filter::NEAREST)
            .min_filter(vk::Filter::NEAREST)
            .address_mode_u(vk::SamplerAddressMode::CLAMP_TO_EDGE)
            .address_mode_v(vk::SamplerAddressMode::CLAMP_TO_EDGE)
            .address_mode_w(vk::SamplerAddressMode::CLAMP_TO_EDGE)
            .mipmap_mode(vk::SamplerMipmapMode::NEAREST);
        self.hud_font_sampler = unsafe {
            self.device
                .create_sampler(&sampler_info, None)
                .map_err(|e| format!("创建 HUD 字体采样器失败: {}", e))?
        };
        // 描述符集布局（binding 0=纹理+采样器）
        let bindings = [
            vk::DescriptorSetLayoutBinding::default()
                .binding(0)
                .descriptor_type(vk::DescriptorType::COMBINED_IMAGE_SAMPLER)
                .descriptor_count(1)
                .stage_flags(vk::ShaderStageFlags::FRAGMENT),
        ];
        let dsl_info = vk::DescriptorSetLayoutCreateInfo::default().bindings(&bindings);
        self.hud_font_descriptor_layout = unsafe {
            self.device
                .create_descriptor_set_layout(&dsl_info, None)
                .map_err(|e| format!("创建 HUD 字体描述符布局失败: {}", e))?
        };
        // 分配描述符集并写入
        let pool_sizes = [vk::DescriptorPoolSize::default()
            .ty(vk::DescriptorType::COMBINED_IMAGE_SAMPLER)
            .descriptor_count(1)];
        let pool_info = vk::DescriptorPoolCreateInfo::default()
            .pool_sizes(&pool_sizes)
            .max_sets(1);
        let pool = unsafe {
            self.device
                .create_descriptor_pool(&pool_info, None)
                .map_err(|e| format!("创建 HUD 字体描述符池失败: {}", e))?
        };
        self.hud_font_pool = pool;
        let alloc_info = vk::DescriptorSetAllocateInfo::default()
            .descriptor_pool(pool)
            .set_layouts(&[self.hud_font_descriptor_layout]);
        self.hud_font_descriptor_set = unsafe {
            self.device
                .allocate_descriptor_sets(&alloc_info)
                .map_err(|e| format!("分配 HUD 字体描述符集失败: {}", e))?
            .remove(0)
        };
        let font_info = vk::DescriptorImageInfo::default()
            .image_layout(vk::ImageLayout::SHADER_READ_ONLY_OPTIMAL)
            .image_view(self.hud_font_image_view)
            .sampler(self.hud_font_sampler);
        let write = [vk::WriteDescriptorSet::default()
            .dst_set(self.hud_font_descriptor_set)
            .dst_binding(0)
            .descriptor_type(vk::DescriptorType::COMBINED_IMAGE_SAMPLER)
            .image_info(&[font_info])];
        unsafe {
            self.device.update_descriptor_sets(&write, &[]);
        }
        log::info!("HUD 字体图集就绪: {}x{} R8G8B8A8", FONT_ATLAS_SIZE, FONT_ATLAS_SIZE);
